<script setup>
import { computed, defineProps, defineEmits } from "vue";
import chevron_down from "../assets/icons/chevron_down.svg";
import chevron_left from "../assets/icons/chevron_left.svg";
import chevron_right from "../assets/icons/chevron_right.svg";
const btnProps = defineProps({
    text: String,
    type: {
        type: String,
        default: "button",
    },
    disabled: Boolean,
    size: String,
    arrow: {
        type: Boolean,
        default: false
    },
    arrow_direction: String,
    icon: String,
    customClass: String,
    leftIconClass: String,
    tooltip: String,
    btnBackgroundType: {
        type: String,
        default: "emphasized"
    }
});
const emit = defineEmits(["button-click"]);
const btnIcon = btnProps.icon ? new URL(btnProps.icon, import.meta.url).href : null;
var arrowIcon = "";
if(btnProps.arrow_direction === "down") {
    arrowIcon = chevron_down;
} else if(btnProps.arrow_direction === "left") {
    arrowIcon = chevron_left;
} else if(btnProps.arrow_direction === "right") {
    arrowIcon = chevron_right;
}
// Reactive, not a one-time mutation of the (read-only) disabled prop at mount - the button
// must go back to its real background type, and re-enable, whenever `disabled` later
// becomes false (e.g. a form drawer that opens disabled and enables once required fields
// are filled). Mutating btnProps.btnBackgroundType directly only ran once at setup, so a
// button that started disabled stayed permanently disabled even after `disabled` flipped.
const effectiveBtnBackgroundType = computed(() =>
    btnProps.disabled ? "neutral" : btnProps.btnBackgroundType
);

/**
 *
 */
function handleClick() {
    emit("button-click");
}

const sizeClass = ["small", "medium", "large"].includes(btnProps.size) ? "btn-container" + " " + `btn-${btnProps.size}` : "btn-container";
const arrowSizeClass = ["small", "medium", "large"].includes(btnProps.size) ? "button-arrow-" + `${btnProps.size}` : "";
</script>

<template>
    <button :type="btnProps.type" :class="`${sizeClass} ${effectiveBtnBackgroundType} ${btnProps.customClass || ''}`" @click="handleClick"
        :data-tooltip="btnProps.tooltip" :disabled="btnProps.disabled">
        <img v-if="btnProps.icon" class="icon" :src=btnIcon alt="icon" />
        <span class="button-text">
            <slot>{{ text }}</slot>
        </span>
        <img v-if="btnProps.arrow" :class="`${arrowSizeClass}`" :src=arrowIcon />
    </button>
</template>

<script setup>
import { defineProps, defineEmits, ref, watch, computed } from "vue";

const props = defineProps({
    modelValue: {
        type: String,
        default: ""
    },
    displayValue: {
        type: String,
        default: null
    },
    placeholder: String,
    size: String,
    inputType: {
        type: String,
        default: "text"
    },
    id: {
        type: String,
        default: () => `input-${Math.random().toString(36).slice(2)}`
    },
    visibleColor: {
        type: Boolean,
        default: false
    },
    disabled: {
        default: false,
        type: Boolean
    }
});

const emit = defineEmits(["update:modelValue"]);
const hasUserEdited = ref(false);
const hasExternalValue = ref(false);
const isFocused = ref(false);
const sizeClass = ["small", "large"].includes(props.size)
    ? `input-container input-${props.size}`
    : "input-container";

/**
 * Watch for external value changes (API / programmatic)
 */
watch(
  () => props.modelValue,
  (newVal, oldVal) => {
    if (!hasUserEdited.value && newVal !== "" && newVal !== oldVal) {
      hasExternalValue.value = true;
    }
    if (newVal === "") {
      hasExternalValue.value = false;
      hasUserEdited.value = false;
    }
  }
);

/**
	 * 
	 * @param e 
	 */
function onInput(e) {
    hasUserEdited.value = true;
    hasExternalValue.value = false;
    emit("update:modelValue", e.target.value);
}

/**
 * Border color logic
 *
 * NOTE: these three colors resolve to Figma's "Indicator/Accent 9",
 * "Indicator/Accent 8", and "Neutral/300" — i.e. chart/data-viz accent
 * colors being reused here as input-state indicators (edited vs.
 * externally-set vs. default). There's no dedicated semantic token for
 * "input border - edited" / "input border - external value" in the
 * exported set. Works today because the hex values happen to exist, but
 * it's borrowing meaning from an unrelated token family — worth asking
 * design to add real --color-input-border-* semantic tokens so this
 * isn't silently coupled to the chart palette.
 */
const borderColor = computed(() => {
  if (hasUserEdited.value) return "var(--primitive-color-indicator-accent-9)"; // #C600BF
  if (hasExternalValue.value) return "var(--primitive-color-indicator-accent-8)"; // #F5CC00
  return "var(--primitive-color-neutral-300)"; // #9CA5BE
});
</script>

<template>
    <!-- NOTE: focused-state border was hardcoded to pure black (#000000),
         which has no matching token in the exported set either. Left as a
         literal for now — flag with design whether focus should actually
         use --primitive-color-neutral-700 (#202430, the darkest neutral)
         or a real focus-ring token like Button's --color-button-focus-border. -->
    <div class="mainInputContainer" :style="{ borderColor: isFocused ? '#000000' : 'var(--primitive-color-neutral-300)' }">
        <input 
            :id="id" 
            :type="inputType" 
            :placeholder="placeholder" 
            :class="sizeClass" 
            :value="isFocused ? modelValue : (displayValue ?? modelValue)"
            :disabled="disabled" 
            @input="onInput"
            @focus="isFocused = true"
            @blur="isFocused = false"
            :style="{
                '--border-color': borderColor,
                borderLeft: visibleColor ? `4px solid ${borderColor}` : 'none'
            }"
        />
    </div>
</template>

<!--

  NOTE: this is a new component, not an edit to the existing Select.vue.
  known-issues.md flags Select.vue as having almost no real styling (just
  native <select> chrome) — this is what a properly designed select should
  look like, built to match the Figma "Algorithm" dropdown. Treat this as
  the reference implementation for Select.vue's eventual design pass
  rather than a silent replacement — Select.vue may be used elsewhere in
  the app and swapping it needs a deliberate migration, not a rename.

<script setup>
import { ref } from "vue";

const props = defineProps({
  modelValue: { type: String, default: "" },
  options: { type: Array, required: true }, // array of strings
  label: { type: String, default: "" },
  required: { type: Boolean, default: false },
});
const emit = defineEmits(["update:modelValue"]);

const open = ref(false);
function select(option) {
  emit("update:modelValue", option);
  open.value = false;
}
</script>

<template>
  <div class="field">

        <span>{{ modelValue || "Select..." }}</span>
        <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
          <path d="M6 9l6 6 6-6" />
        </svg>
      </button>
      <ul v-if="open" class="select-menu" role="listbox">
        <li
          v-for="option in options"
          :key="option"
          class="select-option"
          role="option"
          @click="select(option)"
        >
          {{ option }}
        </li>
      </ul>

  </div>
</template>
